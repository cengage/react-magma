import {
  Table,
  TableHead,
  TableRow,
  TableHeaderCell,
  TableBody,
  TableCell,
  Tag,
  TagSize,
  magma,
} from 'react-magma-dom';
import semver from 'semver';

import { getVersions } from '../../utils';

export const NextVersionTable = () => {
  const [tags, setTags] = React.useState([]);
  const [row, setRow] = React.useState();

React.useEffect(() => {
const getVersionsAndTags = new Promise(res => res(getVersions()));

    getVersionsAndTags.then(function (val) {
      setTags(val.tags);
    });

}, []);
React.useEffect(() => {
if (tags && tags.next) {
setRow(

<TableRow>
<TableCell width="160px" style={{ whiteSpace: 'nowrap' }}>
  v{tags.next.version}
</TableCell>
<TableCell>
<a href="https://next--upbeat-sinoussi-f675aa.netlify.app/">Docs</a>
</TableCell>
</TableRow>
);
} else {
setRow(<></>);
}
}, [tags]);

return (

<>
  <Table minWidth={100}>
    <TableHead>
      <TableRow>
        <TableHeaderCell width="160px">Version</TableHeaderCell>
        <TableHeaderCell>Docs link</TableHeaderCell>
      </TableRow>
    </TableHead>
    <TableBody>{row}</TableBody>
  </Table>
</>
); };

export const ScrollableTableContainer = ({ children }) => {
  const [showScrollHint, setShowScrollHint] = React.useState(false);
  const tableContainerRef = React.useRef(null);

  React.useEffect(() => {
    const scroller = tableContainerRef.current?.querySelector('table')?.parentElement;

    if (!scroller) return;

    const updateScrollHint = () => {
      const hasMoreContent =
        scroller.scrollLeft + scroller.clientWidth < scroller.scrollWidth - 1;

      setShowScrollHint(hasMoreContent);
    };

    updateScrollHint();
    scroller.addEventListener('scroll', updateScrollHint);
    window.addEventListener('resize', updateScrollHint);

    return () => {
      scroller.removeEventListener('scroll', updateScrollHint);
      window.removeEventListener('resize', updateScrollHint);
    };
  }, [children]);

  return (
    <div ref={tableContainerRef} style={{ position: 'relative' }}>
      {children}
      {showScrollHint ? (
        <div
          aria-hidden="true"
          style={{
            background: `linear-gradient(to right, transparent, ${magma.colors.neutral0})`,
            bottom: 0,
            pointerEvents: 'none',
            position: 'absolute',
            right: 0,
            top: 0,
            width: '48px',
          }}
        />
      ) : null}
    </div>
  );
};

export const ReleasedVersionsTable = () => {
  const [allVersions, setAllVersions] = React.useState([]);
  const [rows, setRows] = React.useState();

React.useEffect(() => {
const getVersionsAndTags = new Promise(res => res(getVersions()));

    getVersionsAndTags.then(function (val) {
      setAllVersions(val.versions.reverse());
    });

}, []);
React.useEffect(() => {
if (allVersions) {
const tableRow = allVersions.map((row, i) => {
return (

<TableRow key={`row${i}`}>
  <TableCell
    key={`row${i}-version`}
    width="140px"
    style={{ whiteSpace: 'nowrap' }}
  >
    v{row.version}
    {row.tags.includes('latest') || i === 0 ? <span>&nbsp;✅&nbsp;</span> : ''}
    &nbsp;
    {row.deprecated ? <Tag size={TagSize.small}>Deprecated</Tag> : null}
  </TableCell>
  <TableCell key={`row${i}-date`}>{row.date}</TableCell>
  <TableCell key={`row${i}-docs`}>
    {semver.gte(row.version, '4.0.0') ||
    row.version === '3.11.0' ||
    row.version === '2.6.2' ? (
      <a href={`https://react-magma.cengage.com/version/${row.version}`}>
        Docs
      </a>
    ) : (
      <></>
    )}
  </TableCell>
  <TableCell key={`row${i}-react`}>{row.peerDependencies?.react}</TableCell>
</TableRow>
); });

      setRows(tableRow);
    } else {
      setRows(<></>);
    }

}, [allVersions]);

return (

<>
  <ScrollableTableContainer>
    <Table minWidth={760} style={{ tableLayout: 'fixed', width: '760px' }}>
      <TableHead>
        <TableRow>
          <TableHeaderCell width="140px">Version</TableHeaderCell>
          <TableHeaderCell width="190px">Release date</TableHeaderCell>
          <TableHeaderCell width="150px">Docs link</TableHeaderCell>
          <TableHeaderCell width="280px">Supported React Version</TableHeaderCell>
        </TableRow>
      </TableHead>
      <TableBody>{rows}</TableBody>
    </Table>
  </ScrollableTableContainer>
</>
); };
